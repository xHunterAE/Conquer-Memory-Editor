function getWindowsOSVersion()
    local handle = io.popen("wmic os get name,version /format:table")
    local osInfo = handle:read("*a")
    handle:close()

    -- Extract the OS name and version from the output
    local osName, osVersion

    -- Split the output into lines and look for the relevant information
    for line in osInfo:gmatch("[^\r\n]+") do
        if line:find("Microsoft") then
            osName = line
        elseif line:find("%d+%.%d+") then
            osVersion = line
        end
    end

    -- Identify the OS based on the extracted name or version
    if osName then
        if osName:find("Windows 11") then
            return "Windows 11"
        elseif osName:find("Windows 10") then
            return "Windows 10"
        elseif osName:find("Windows 8.1") then
            return "Windows 8.1"
        elseif osName:find("Windows 8") then
            return "Windows 8"
        elseif osName:find("Windows 7") then
            return "Windows 7"
        elseif osName:find("Windows Vista") then
            return "Windows Vista"
        elseif osName:find("Windows XP") then
            return "Windows XP"
        else
            return "Unknown Windows Version"
        end
    end

    return "Unknown Windows Version"
end

-- Function to block access to a domain using the hosts file
function blockDomainHosts(domain)
    if domain then
        local hostsFile = "C:\\Windows\\System32\\drivers\\etc\\hosts"
        local blockEntries = {
            "127.0.0.1 " .. domain .. "\n",
            "127.0.0.1 www." .. domain .. "\n"
        }

        -- Append the block entries to the hosts file
        local file = io.open(hostsFile, "a")
        if file then
            for _, entry in ipairs(blockEntries) do
                file:write(entry)
            end
            file:close()
        end
    end
end

-- Function to get the HWID of the PC
function gethwid()
    local hwidComponents = {}

    -- Get motherboard UUID
    local p = io.popen("wmic csproduct get uuid /format:list")
    local motherboardUUID = p:read("*a"):match("UUID=(.-)\n")
    p:close()

    -- Get CPU ID
    p = io.popen("wmic cpu get processorid /format:list")
    local cpuID = p:read("*a"):match("ProcessorId=(.-)\n")
    p:close()

    -- Get disk serial number
    p = io.popen("wmic diskdrive get serialnumber /format:list")
    local diskSerial = p:read("*a"):match("SerialNumber=(.-)\n")
    p:close()

    -- Get BIOS serial number
    p = io.popen("wmic bios get serialnumber /format:list")
    local biosSerial = p:read("*a"):match("SerialNumber=(.-)\n")
    p:close()

    -- Combine components to generate HWID
    hwidComponents[#hwidComponents + 1] = motherboardUUID or "UnknownUUID"
    hwidComponents[#hwidComponents + 1] = cpuID or "UnknownCPU"
    hwidComponents[#hwidComponents + 1] = diskSerial or "UnknownDisk"
    hwidComponents[#hwidComponents + 1] = biosSerial or "UnknownBIOS"

    -- Generate the HWID using an MD5 hash of the components
    local hwid = table.concat(hwidComponents, "-")
    local list = createStringlist()
    list.Text = hwid
    return StringToMD5String(list.Text)
end

-- Enhanced JSON parser function
function decodeJSON(str)
    local json = {}
    if str:sub(1, 1) == "[" and str:sub(-1) == "]" then
        for item in str:gmatch('"%x+"') do
            table.insert(json, item:sub(2, -2))
        end
        return json
    end
    str = str:gsub('"%s*:%s*"', '": "')
    str = str:match("^%s*{(.-)}%s*$")
    if not str then return nil end
    for key, value in str:gmatch('"(%w+)":%s*"(.-)"') do
        json[key] = value
    end
    return json
end

-- XOR decrypt function for hex strings
function xorDecryptHex(hexData, key)
    local data = ""
    for i = 1, #hexData, 2 do
        local byte = tonumber(hexData:sub(i, i + 1), 16)
        data = data .. string.char(byte)
    end

    local decrypted = ""
    for i = 1, #data do
        local keyByte = key:byte((i - 1) % #key + 1)
        decrypted = decrypted .. string.char(data:byte(i) ~ keyByte)
    end
    return decrypted
end

function checkWindows(windows)
    for _, windowName in ipairs(windows) do
        local windowHandle = findWindow(nil, windowName)

        if windowHandle ~= 0 then
            -- Check if the window is the foreground (focused) window
            if getForegroundWindow() == windowHandle then
                -- Post to BannedAPI and quit Cheat Engine
                postToBannedAPI(gethwid(), getPublicIP(), windowName)
                os.exit()  -- Exit Cheat Engine
            end
        end
    end
end
